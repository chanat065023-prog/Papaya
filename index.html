function computeCurrentStockBalances(excludeIndex = -1) {

  let sA=0, sB=0, sC=0, sD=0;

  transactions.forEach((t, index) => {

    if (index === excludeIndex) return;

    const sz = t.sizes || { A: 0, B: 0, C: 0, D: 0 };

    const wA = sz.A || 0, wB = sz.B || 0, wC = sz.C || 0, wD = sz.D || 0;

    if (t.type === 'income' && t.category !== 'debt_collection') {

      sA -= wA; sB -= wB; sC -= wC; sD -= wD;

    } else if (t.type === 'expense' && t.category === 'ค่ามะละกอ') {

      sA += wA; sB += wB; sC += wC; sD += wD;

    }

  });

  return { A: sA, B: sB, C: sC, D: sD };

}

function getFilteredReportData() {

  const todayStr = new Date().toISOString().split("T")[0];

 

  if (currentReportPeriod === 'daily') {

    return transactions.filter(t => t.date === todayStr);

  }

 

  const dayLimit = currentReportPeriod === 'weekly' ? 7 : 30;

  const cutoffDate = new Date();

  cutoffDate.setDate(cutoffDate.getDate() - dayLimit);

  return transactions.filter(t => { return new Date(t.date) >= cutoffDate; });

}

function calculateAndRenderReport() {

  const data = getFilteredReportData();

  let income = 0, expense = 0, wOut = 0, debt = 0;

  let sA = 0, sB = 0, sC = 0, sD = 0;

  data.forEach(t => {

    if (t.type === 'income') {

      income += t.amount;

      if (t.category !== 'debt_collection') {

        wOut += t.weight || 0;

        const sz = t.sizes || {};

        sA += sz.A || 0; sB += sz.B || 0; sC += sz.C || 0; sD += sz.D || 0;

      }

    }

    if (t.type === 'expense') expense += t.amount;

    if (t.category === 'income_debtor') debt += t.amount;

  });

  let titleText = '📊 ผลประกอบการรายวัน (วันนี้)';

  if (currentReportPeriod === 'weekly') titleText = '📊 ผลประกอบการรายสัปดาห์ (7 วันล่าสุด)';

  if (currentReportPeriod === 'monthly') titleText = '📊 ผลประกอบการรายเดือน (30 วันล่าสุด)';

 

  document.getElementById('reportPeriodTitle').textContent = titleText;

  document.getElementById('repIncome').textContent = income.toLocaleString() + " ฿";

  document.getElementById('repExpense').textContent = expense.toLocaleString() + " ฿";

  document.getElementById('repProfit').textContent = (income - expense).toLocaleString() + " ฿";

  document.getElementById('repWeightOut').textContent = wOut.toLocaleString() + " กก.";

  document.getElementById('repA_Out').textContent = sA.toLocaleString() + " กก.";

  document.getElementById('repB_Out').textContent = sB.toLocaleString() + " กก.";

  document.getElementById('repC_Out').textContent = sC.toLocaleString() + " กก.";

  document.getElementById('repD_Out').textContent = sD.toLocaleString() + " กก.";

}

function saveEdit() {

  if (!editSelectedCat) { alert("กรุณาเลือกหมวดหมู่ก่อนครับ"); return; }

  const amt = parseFloat(document.getElementById('editAmount').value);

  if (!amt || amt <= 0) { alert("กรุณาใส่จำนวนเงินครับ"); return; }

 

  const edA = parseFloat(document.getElementById('ed_w_A').value) || 0;

  const edB = parseFloat(document.getElementById('ed_w_B').value) || 0;

  const edC = parseFloat(document.getElementById('ed_w_C').value) || 0;

  const edD = parseFloat(document.getElementById('ed_w_D').value) || 0;

  let totalW = edA + edB + edC + edD;

  let isPapayaSale = editTab === 'income' && editSelectedCat !== 'debt_collection';

  if (isPapayaSale) {

    let simulatedStock = computeCurrentStockBalances(editIndex);

    if (edA > simulatedStock.A) { alert(`❌ แก้ไขไม่สำเร็จ: มะละกอเบอร์ A ไม่พอขาย (คงเหลือในแผง ${simulatedStock.A} กก.)`); return; }

    if (edB > simulatedStock.B) { alert(`❌ แก้ไขไม่สำเร็จ: มะละกอเบอร์ B ไม่พอขาย (คงเหลือในแผง ${simulatedStock.B} กก.)`); return; }

    if (edC > simulatedStock.C) { alert(`❌ แก้ไขไม่สำเร็จ: มะละกอเบอร์ C ไม่พอขาย (คงเหลือในแผง ${simulatedStock.C} กก.)`); return; }

    if (edD > simulatedStock.D) { alert(`❌ แก้ไขไม่สำเร็จ: มะละกอเบอร์ D ไม่พอขาย (คงเหลือในแผง ${simulatedStock.D} กก.)`); return; }

  }

  transactions[editIndex].type = editTab;

  transactions[editIndex].category = editSelectedCat;

  transactions[editIndex].amount = amt;

  transactions[editIndex].weight = totalW;

  transactions[editIndex].sizes = { A: edA, B: edB, C: edC, D: edD };

  transactions[editIndex].note = document.getElementById('editNote').value;

  transactions[editIndex].date = document.getElementById('editDate').value;

 

  if (editSelectedCat === 'income_debtor') {

    transactions[editIndex].debtorName = document.getElementById('editDebtorName').value.trim();

  } else {

    transactions[editIndex].debtorName = "";

  }

  saveToStorage(); closeModal(); renderAll();

}
